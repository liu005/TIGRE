%% Cargo DR simulation only: forward projection, no 3D reconstruction.
% attenuationPath: optional MAT file with a 3D xyz array named attenuation,
% containing finite nonnegative linear attenuation coefficients in 1/mm.
geometryPath='__SAVED_GEOMETRY__';
attenuationPath='';
uniformMu=0.001;
voxelCounts=[64;128;64]; % Small simulation grid; physical size is preserved.
outputPath=fullfile(fileparts(mfilename('fullpath')),'dr_results.mat');
document=geometryFile('load',geometryPath); geo=document.geo; surface=document.surface;
if isempty(attenuationPath)
    validateattributes(uniformMu,{'numeric'},{'scalar','finite','nonnegative'});
    validateattributes(voxelCounts,{'numeric'},{'numel',3,'integer','positive'});
    attenuation=repmat(single(uniformMu),voxelCounts');
else
    data=load(attenuationPath,'attenuation'); attenuation=single(data.attenuation);
    voxelCounts=size(attenuation)';
end
assert(ndims(attenuation)==3 && all(isfinite(attenuation(:))) && all(attenuation(:)>=0), ...
    'TIGRE:dr:attenuation','Supply finite nonnegative 3D xyz attenuation in 1/mm');
geo.nVoxel=voxelCounts(:); geo.sVoxel=geo.sVoxel(:); geo.dVoxel=geo.sVoxel./geo.nVoxel;
[surface,geo]=fixedObjectGeometry(surface,geo);
validateCTTransmission(surface,geo);
projections=Ax_surface(attenuation,geo,surface);
transmission=exp(-projections); % Normalized monoenergetic transmission.
module_slices=surface.module_slices; dr_image=[];
for module=1:size(module_slices,1)
    bounds=module_slices(module,:); rows=bounds(1)+1:bounds(2);
    column=floor((bounds(3)+bounds(4)-1)/2)+1;
    image=reshape(projections(rows,column,:),numel(rows),[]);
    dr_image=[dr_image;image]; %#ok<AGROW>
end
valid_mask=surface.valid_mask; view_times_s=[];
if isfield(geo,'view_times_s'), view_times_s=geo.view_times_s; end
save(outputPath,'projections','transmission','dr_image','module_slices','valid_mask','view_times_s','geometryPath','geo');
fprintf('Saved DR projections: %s (no reconstruction)\n',outputPath);
